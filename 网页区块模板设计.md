```html
<!-- 导航栏：Logo图片 + 网站名称 + 菜单 + CTA按钮 -->
<section class="border-b border-gray-100 bg-white/80 backdrop-blur-sm sticky top-0 z-20">
  <div class="max-w-7xl mx-auto px-4 sm:px-6 lg:px-8">
    <div class="flex items-center justify-between h-16 md:h-20">
      <!-- Logo 区域：图片 + 品牌名称 -->
      <div class="flex items-center gap-3 shrink-0">
        <img class="h-8 w-auto rounded-lg shadow-sm" src="{{logo}}" alt="{{brandName}} Logo" onerror="this.src='https://placehold.co/32x32/4F46E5/white?text=Logo';">
        <span class="text-xl font-semibold text-gray-800 tracking-tight">{{brandName}}</span>
      </div>

      <!-- 导航菜单区（支持二级菜单） -->
      <nav class="hidden md:flex items-center space-x-1 lg:space-x-2">
        {{#each navLinks}}
        <div class="relative group">
          {{#if children}}
          <!-- 带二级菜单的一级菜单项 -->
          <button class="flex items-center gap-1 px-3 py-2 text-gray-600 hover:text-indigo-600 text-sm font-medium transition-colors rounded-lg hover:bg-gray-50">
            {{text}}
            <svg class="w-4 h-4 transition-transform group-hover:rotate-180" fill="none" stroke="currentColor" viewBox="0 0 24 24">
              <path stroke-linecap="round" stroke-linejoin="round" stroke-width="2" d="M19 9l-7 7-7-7"></path>
            </svg>
          </button>
          <!-- 二级菜单 -->
          <div class="absolute left-0 mt-1 w-56 bg-white rounded-xl shadow-lg border border-gray-100 opacity-0 invisible group-hover:opacity-100 group-hover:visible transition-all duration-200 z-30">
            <div class="py-2">
              {{#each children}}
              <a href="{{link}}" class="flex items-center gap-3 px-4 py-2.5 text-sm text-gray-600 hover:bg-indigo-50 hover:text-indigo-600 transition" {{#if newWindow}}target="_blank" rel="noopener noreferrer"{{/if}}>
                {{#if icon}}<span class="text-indigo-500">{{icon}}</span>{{/if}}
                {{text}}
              </a>
              {{/each}}
            </div>
          </div>
          {{else}}
          <!-- 无二级菜单的一级菜单项（直接链接） -->
          <a href="{{link}}" class="px-3 py-2 text-gray-600 hover:text-indigo-600 text-sm font-medium transition-colors rounded-lg hover:bg-gray-50" {{#if newWindow}}target="_blank" rel="noopener noreferrer"{{/if}}>{{text}}</a>
          {{/if}}
        </div>
        {{/each}}
      </nav>

      <!-- 右侧 CTA 按钮 + 移动端菜单图标 -->
      <div class="flex items-center gap-3">
        {{#each buttons}}
        <a href="{{link}}" class="hidden sm:inline-flex items-center justify-center {{#if isPrimary}}bg-indigo-600 hover:bg-indigo-700 text-white shadow-sm{{else}}border border-gray-300 text-gray-700 hover:bg-gray-50{{/if}} text-sm font-medium px-5 py-2 rounded-full transition-colors" {{#if newWindow}}target="_blank" rel="noopener noreferrer"{{/if}}>{{text}}</a>
        {{/each}}
        <!-- 移动端菜单占位（仅展示，无JS交互） -->
        <button class="md:hidden p-2 rounded-lg text-gray-500 hover:bg-gray-100">
          <svg class="w-6 h-6" fill="none" stroke="currentColor" viewBox="0 0 24 24">
            <path stroke-linecap="round" stroke-linejoin="round" stroke-width="2" d="M4 6h16M4 12h16M4 18h16"></path>
          </svg>
        </button>
      </div>
    </div>
  </div>
</section>
```
